### 1.5 `sprite_card_connect_curve.vert / .frag` —— 连线本体（**最有价值的部分**）

**vert**：把一条矩形条带按曲线弯折。

```glsl
float t = clamp(a_texCoord.x, 0.0, 1.0);   // t = 沿曲线参数（0..1）
if (u_use_bezier > 0.5) {
    // 三次贝塞尔：p0 = u_origin, p1 = u_control1, p2 = u_control2, p3 = u_end_point
    curve   = cubicBezier(...);
    tangent = cubicBezierTangent(...);
    normal  = vec2(-tangent.y, tangent.x) / max(length(tangent), 0.0001);
    pos.xy  = curve + normal * (a_texCoord.y - 0.5) * 2.0 * u_half_thickness;   // 沿法线撑开厚度
} else {
    // 抛物线退化模式：拱高 u_curve_height、幂 u_curve_power、横向偏移 u_curve_side
    float arch = 4.0 * t * (1.0 - t);
    float shaped = pow(max(arch, 0.0), max(u_curve_power, 0.001));
    pos.y += shaped * u_curve_height;
    pos.x += (t - 0.5) * shaped * u_curve_side;
}
```

**frag**：让贴图沿条带**无缝滚动**（这是整套东西的灵魂）。

```glsl
vec2 getFlowUV(float phase_x, float phase_v, float texel_x, float texel_y) {
    vec2 uv = v_texCoord;
    uv.x = mix(texel_x, 1.0 - texel_x, phase_x);                       // 横向循环滚动 + 去边距
    uv.y = fract(v_texCoord.y + sin((phase_v + u_time * 0.35) * 6.28318) * u_flow_force); // 纵向轻微波动
    uv.y = mix(texel_y, 1.0 - texel_y, uv.y);                          // 去边距
    return uv;
}
// 双采样消接缝：a、b 相差半周期，按到接缝的距离加权混合 —— 这样滚动时看不到接缝
float phase   = fract(v_texCoord.x + u_time * u_flow_speed);
float phase_b = fract(phase + 0.5);
vec4  a = sampleFlow(phase,   phase,   ...);
vec4  b = sampleFlow(phase_b, phase,   ...);
float d_a = min(phase, 1.0 - phase);
float d_b = abs(phase - 0.5);
float w   = d_a / (d_a + d_b + 0.0001);
vec4  tex = mix(a, b, w);
```

> **`texel_x = 1.5 / 1028.0`、`texel_y = 1.5 / 260.0` 这两个常量证明了它的配对贴图
> 就是 `line_connect_1.png`（1024×256，本工程实测尺寸）**，两者同源。
> 也就是说：**`ui_trail\line_connect_1.png` + 这对 shader = 原版卡牌之间的那条流动光带**。

`line_connect_1.png` 实测：1024×256，内容是一条**左右两端淡出、中间带柔和水波纹**的横向白色光带。

## 5.4 连线 shader 移植（1:1，可直接落地）


**`res://CznHeidemarie/link_vfx/card_connect_flow.gdshader`**（由 vert + frag 合并成一个）

```glsl
// 「卡牌连接」光带着色器 —— 由 CNZ 原版 sprite_card_connect_curve.vert/.frag 移植
// 顶点：把一条矩形网格按贝塞尔/抛物线弯折；片元：让 line_connect_1.png 沿条带无缝滚动
shader_type canvas_item;
render_mode blend_add, unshaded;   // 原版为加算混合

uniform bool  u_use_bezier = true;          // true=贝塞尔（原版主模式）
uniform vec2  u_origin     = vec2(0.0);     // 起点
uniform vec2  u_control1   = vec2(0.0);     // 控制点 1
uniform vec2  u_control2   = vec2(0.0);     // 控制点 2
uniform vec2  u_end_point  = vec2(0.0);     // 终点
uniform float u_half_thickness = 40.0;      // 半厚度（原版按曲线法线撑开）

uniform float u_arc_height = 120.0;         // 抛物线模式：拱高
uniform float u_arc_power  = 1.0;           // 抛物线模式：幂次
uniform float u_arc_side   = 0.0;           // 抛物线模式：横向偏移

uniform float u_flow_speed  = 0.35;         // 沿曲线滚动速度（原版 u_flow_speed）
uniform float u_flow_force  = 0.02;         // 纵向波动幅度（原版 u_flow_force）
uniform vec3  u_tint_color  = vec3(0.0);    // 染色调（原版 u_tint_color，青蓝系）
uniform float u_tint_strength = 0.0;
uniform float u_alpha_mul     = 1.0;
// 贴图 1024x256；原版写死 1.5/1028 与 1.5/260（1.5 为防溢色边距）
uniform float u_texel_x = 1.5 / 1024.0;
uniform float u_texel_y = 1.5 / 256.0;

// ---- 原版 cubicBezier / cubicBezierTangent 原样保留 ----
vec2 cubic_bezier(vec2 p0, vec2 p1, vec2 p2, vec2 p3, float t) {
    float mt = 1.0 - t;
    return mt * mt * mt * p0 + 3.0 * mt * mt * t * p1 + 3.0 * mt * t * t * p2 + t * t * t * p3;
}
vec2 cubic_bezier_tangent(vec2 p0, vec2 p1, vec2 p2, vec2 p3, float t) {
    float mt = 1.0 - t;
    return 3.0 * mt * mt * (p1 - p0) + 6.0 * mt * t * (p2 - p1) + 3.0 * t * t * (p3 - p2);
}

void vertex() {
    float t = clamp(UV.x, 0.0, 1.0);        // 网格 UV.x = 沿曲线参数，与原版 a_texCoord.x 同义
    if (u_use_bezier) {
        vec2 curve   = cubic_bezier(u_origin, u_control1, u_control2, u_end_point, t);
        vec2 tangent = cubic_bezier_tangent(u_origin, u_control1, u_control2, u_end_point, t);
        vec2 normal  = vec2(-tangent.y, tangent.x) / max(length(tangent), 0.0001);
        // UV.y = 0/1 → 沿法线撑开 ±u_half_thickness
        VERTEX = curve + normal * (UV.y - 0.5) * 2.0 * u_half_thickness;
    } else {
        float arch   = 4.0 * t * (1.0 - t);
        float shaped = pow(max(arch, 0.0), max(u_arc_power, 0.001));
        vec2  pos    = VERTEX;
        pos.y += shaped * u_arc_height;
        pos.x += (t - 0.5) * shaped * u_arc_side;
        VERTEX = pos;
    }
}

// ---- 原版 getFlowUV / sampleFlow：双采样消接缝 ----
vec2 flow_uv(float phase_x, float phase_v) {
    vec2 uv = UV;
    uv.x = mix(u_texel_x, 1.0 - u_texel_x, phase_x);
    uv.y = fract(uv.y + sin((phase_v + TIME * 0.35) * 6.28318) * u_flow_force);
    uv.y = mix(u_texel_y, 1.0 - u_texel_y, uv.y);
    return uv;
}

void fragment() {
    float phase   = fract(UV.x + TIME * u_flow_speed);
    float phase_b = fract(phase + 0.5);
    vec4  a = texture(TEXTURE, flow_uv(phase,   phase));
    vec4  b = texture(TEXTURE, flow_uv(phase_b, phase));
    float d_a = min(phase, 1.0 - phase);
    float d_b = abs(phase - 0.5);
    float w   = d_a / (d_a + d_b + 0.0001);   // 到接缝越近，权重越低 → 另一个采样补位
    vec4  tex = mix(a, b, w);

    vec3 tinted = mix(tex.rgb, tex.rgb + u_tint_color, u_tint_strength);
    COLOR = vec4(tinted, tex.a * u_alpha_mul) * COLOR;   // COLOR 保留节点 modulate
}
```

### 9.3 实现上的两处关键处理

1. **shader 里不能用内置 `TEXTURE`**：光带挂在 `MeshInstance2D` 上，节点没有「自带贴图」，
   `TEXTURE` 采样不到东西。改为声明 `uniform sampler2D u_texture : repeat_enable, filter_linear;`，
   由 C# 用 `ShaderMaterial.SetShaderParameter("u_texture", ...)` 绑定
   （这是与原版 Cocos 写法的唯一实质性偏差）。
2. **三角条带网格**：2×(N+1) 顶点、`UV.x` = 曲线参数 `t`、`UV.y` = 0/1，32 段细分。
   曲线**由 C# 每帧求值并写入网格顶点**（`FillRibbonGeometry` + `Bezier` / `BezierTangent`），
   着色器只做片元流动 —— 不再沿用原版「顶点全 0 + shader 求值」的路线，原因见 §9.7 第 1 条。
   着色器里保留了原版的贝塞尔求值代码（`u_use_bezier = true` 可切回），默认关闭。

### 9.8 第二轮实机修复：形态全错（2026-10-15）

第二轮实机截图显示：光带确实出现了，但是**一条粗大的纯白弧**，从卡牌**中部右侧**冒出来向右下延伸；同时**完全看不到卡牌发光框**。

查 `SlayTheSpire2_source/scenes/` 下的场景真值，去掉了猜测：

| 对象 | 真值（场景文件） |
|---|---|
| `hand_card_holder.tscn` 根 `Control` | **`Size = (0,0)`**（`anchors_preset = 0`，无任何 `offset_*`） |
| 其 `Hitbox` | `offset -150..150 / -211..211` → **围绕原点对称** |
| 其 `Flash` | `offset -179..191 / -342..366` → 同样居中原点 |
| `card.tscn` 根 `Control` | 同样 `Size = (0,0)` |
| 其 `Frame` / `Shadow` | `-150..150 / -211..211` → 同样围绕原点居中 |

**结论：手牌 holder 的「原点」就是卡牌中心，卡牌局部可视尺寸约 `300×422`**
（正好等于 `Hitbox.Size`；手牌缩放 `0.8` ⇒ 屏幕上 `240×338`）。

首版代码却按「原点 = 左上角」写，取 `toCanvas * (size * 0.5)` 当卡牌中心 ——
等于**在真正的中心之上又往右下偏了半个卡牌**，于是光带起点落在卡牌中部右侧、画出一条歪掉的白弧；发光框同理整体偏移，落到卡牌外所以看不见。

| # | 问题 | 修复 |
|---|---|---|
| 1 | 锚点偏了整整半个卡牌 | 改为「卡牌中心 = 节点原点」(`toCanvas * Vector2.Zero`)、「顶边中点 = 原点沿局部 −y 半卡高」 |
| 2 | 光带尺寸没乘手牌缩放 | 新增 `GetCanvasSize()`（局部尺寸 × `toCanvas.Scale`）；厚度 / 拱高 / 发光 / 光柱 / 粒子统一按**屏幕尺寸**算 |
| 3 | 光带过粗、加算过曝成白块 | `RibbonHalfThicknessRatio` `0.075 → 0.04`；`u_alpha_mul` `0.9 → 0.55`；新增 `u_tint_color` 青蓝染色（`(0.25,0.75,1)`，强度 `0.5`） |
| 4 | 卡牌发光框看不见 | 随锚点一并归位；外层辉光 `grad2` 额外放大 `GlowInnerScale = 1.18` |
| 5 | `holder.Size` 恒为 0 | `GetCardSize()` 的兜底链本就覆盖（`Hitbox.Size` 即 300×422），只是首版把它的**语义**用错了 |

> 参考图（CNZ 原版「连接」）的观感是：**卡牌整体泛蓝白发光 + 卡牌上方蓝色光晕 + 周围粒子**，牌与牌之间**没有**连线。
> 本工程的光带属于额外的演绎，因此第二轮把它压细压淡，让主视觉落回「卡牌发光」。

### 9.9 第三轮：只保留光带与环绕粒子（2026-10-15）

第三轮实机图确认几何已经正确（光带落在卡牌顶边上方），但还有两个问题：
**光带太粗太白**（横跨整手牌的一条白弧），以及**卡形发光框把卡面盖住**。

按反馈定稿：**移除卡形发光框与光柱，只保留光带 + 环绕光带的粒子。**

| 动作 | 内容 |
|---|---|
| 移除 | `grad1` / `grad2`（卡形发光框）、`ray`（光柱）—— C# 侧 `Glow` 类、`MakeGlowSprite`、`Animate` 及相关分支全部删除 |
| 移除 | 常量 `GlowTextureWidth/Height`、`RayTextureWidth`、`BreathSeconds`、`GlowAlphaHigh/Low`、`RayScaleHigh/Low`、`RayCenterRatio`、`GlowInnerScale`、`ParticleOffsetRatio` |
| 保留 | 资源文件 `card_connect_grad1/grad2/ray.png` **不删**，仍在 `link_vfx/` 里备用（将来想加回发光不必重新切图） |
| 光带 | `RibbonHalfThicknessRatio` `0.04 → 0.03`；改用**节点 modulate 染色** `RibbonTint = (0.48,0.78,1)` 压成青蓝 |
| 粒子 | 从「每张牌一簇」改为「**每条光带一簇**」：发射器挪到曲线 `t = 0.5` 处，`EmissionRectExtents.X` 每帧按光带跨度写入，`LocalCoords = true` 让粒子随光带移动；参数相应放宽（散射 35°、速度 12–42、数量 40），观感是「沿光带升腾的青色碎光」 |

> **为什么之前 shader 里的 `u_tint_color` 没能去掉白色**：那行是 `mix(tex.rgb, tex.rgb + u_tint_color, u_tint_strength)`
> —— **加法**染色，白色 + 蓝色只会更白。要去白必须**乘**，所以改在节点 `Modulate` 上做（`canvas_item` 的 `COLOR` 本就含 modulate，会自动乘进 shader 输出）。
>
> **为什么白弧看着比数值粗**：白贴图 + `blend_add` 在深色背景上会过曝，且 `line_connect_1.png` 自带的辉光很宽，
> 叠加后视觉厚度远超 `RibbonHalfThicknessRatio` 的计算值。染色 + 降到 `0.03` 后才收敛成细弧。

| 链条 | 手牌顺序里带「连接」的牌两两相邻成链；**光带以「holder 对」为键常驻复用**，不每帧重建 |
| 出现 / 消失 | 每条带有 0..1 权重，0.12s 淡入 / 0.18s 淡出，按权重缩放线宽，所以增删卡片不会有跳变 |
| 拖拽延展 | `NPlayerHand.StartCardPlay` 前置补丁记下 holder 与它的**槽位索引**（必须在 reparent 之前取），每帧把拖拽中的 holder 插回链条原位 ⇒ 两个端点各自跟手移动，**光带自然被拉长**，且键不变、不重建节点 |
| 橡皮筋 | 跨距超过「一张卡高」后拱高线性收敛到 0.15×、线宽收敛到 0.78× |
| 打出 | 该牌离开链条 ⇒ 连着它的两条带淡出并把端点收进邻牌；同时左右邻牌之间新出现一条带淡入（「重新接上」） |
| 放下（取消） | holder 被放回容器，键不变 ⇒ 光带自己缩回去 |
| 切换卡拖动 | 数字快捷键换一张拖时新旧 holder 的键都不变，过渡无缝 |
| 粒子 | 每条带一簇 `tri_pati`，沿法线小幅上下浮动、上下两侧都有；跟随「额外粒子」开关 |
